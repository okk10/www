from itertools import product
def are_equivalent(f, g, n):
    for row in product((False, True), repeat=n):
        if f(*row) != g(*row):
            return False
    return True
def de_morgan_left(a, b):
    return not (a and b)
def de_morgan_right(a, b):
    return (not a) or (not b)
def de_morgan_second_left(a, b):
    return not (a or b)
def de_morgan_second_right(a, b):
    return (not a) and (not b)
def implies(a, b):
    return (not a) or b
def implication_as_disjunction(a, b):
    return (not a) or b
def contraposition(a, b):
    return implies(not b, not a)
if __name__ == "__main__":
    print(are_equivalent(de_morgan_left, de_morgan_right, 2))
    print(are_equivalent(de_morgan_second_left, de_morgan_second_right, 2))
    print(are_equivalent(implies, implication_as_disjunction, 2))
    print(are_equivalent(implies, contraposition, 2))
