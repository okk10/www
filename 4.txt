def factorial(n):
    if n < 0:
        return None
    result = 1
    for i in range(2, n + 1):
        result *= i
    return result
def arrangements(n, k):
    if n < 0 or k < 0:
        return 0
    if k > n:
        return 0
    result = 1
    for i in range(k):
        result *= (n - i)
    return result
def combinations(n, k):
    if n < 0 or k < 0:
        return 0
    if k > n:
        return 0
    k = min(k, n - k)
    numerator = 1
    denominator = 1
    for i in range(1, k + 1):
        numerator *= (n - i + 1)
        denominator *= i
    result = numerator // denominator
    return result
print(factorial(0))
print(factorial(5))
print(factorial(-1))
print(arrangements(13, 6))
print(arrangements(4, 2))
print(arrangements(5, 5))
print(arrangements(3, 5))
print(combinations(13, 6))
print(combinations(4, 2))
print(combinations(64, 8))
print(combinations(5, 0))
