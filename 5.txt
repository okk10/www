import random as rnd
def birthday_probability(people):
    if people < 0:
        raise ValueError("Число людей не может быть отрицательным")
    if people <= 1:
        return 0.0
    if people > 365:
        return 1.0
    no_match_probability = 1.0
    for i in range(people):
        no_match_probability *= (365 - i) / 365
    return 1.0 - no_match_probability
def simulate_birthday(people, trials):
    if people < 0:
        raise ValueError("Число людей не может быть отрицательным")
    if trials <= 0:
        raise ValueError("Число испытаний должно быть положительным")
    if people > 365:
        return 1.0
    matches = 0
    for _ in range(trials):
        birthdays = set()
        for _ in range(people):
            day = rnd.randrange(365)
            birthdays.add(day)
        if len(birthdays) < people:
            matches += 1
    return matches / trials
if __name__ == "__main__":
    trials = 100_000
    print(f"{'Людей':>6} {'Точная вероятность':>22} {'Симуляция':>14}")
    for people in range(5, 61, 5):
        exact = birthday_probability(people)
        simulated = simulate_birthday(people, trials)
        print(f"{people:>6} {exact:>22.6f} {simulated:>14.6f}")
    for people in range(1, 367):
        if birthday_probability(people) > 0.5:
            print(f"\nВероятность впервые превышает 0.5 при {people} людях.")
            break
